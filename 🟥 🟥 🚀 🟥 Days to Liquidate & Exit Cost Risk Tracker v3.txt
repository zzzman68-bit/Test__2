//@version=5
indicator("🟥 Days to Liquidate & Exit Cost Risk Tracker v3", overlay=false, max_labels_count=500)

// =====================================================================
// 1. TIMEFRAME GUARDRAIL (CRITICAL FIX)
// =====================================================================
// Prevents inaccurate "Days" math on intraday charts
isIntraday = timeframe.in_seconds() < 86400
if isIntraday
    runtime.error("⚠️ This indicator is designed for Daily (or higher) timeframes. Intraday timeframes will produce inaccurate 'Days to Liquidate' calculations.")

// =====================================================================
// 2. USER & RISK INPUTS
// =====================================================================
myPositionSize     = input.int(100000, title="Your Position Size (Shares/Tokens)", group="Position")
baseParticipation  = input.float(10.0, title="Base Daily Participation Rate (%)", minval=0.1, maxval=100.0, group="Position")
minParticipation   = input.float(0.5, title="Minimum Participation Rate Floor (%)", minval=0.01, maxval=100.0, group="Position")
maxAcceptableDays  = input.int(3, title="Max Acceptable Days to Liquidate", tooltip="Triggers a warning if liquidation takes longer than this threshold.", group="Position")

adtvLength         = input.int(20, title="ADTV Benchmark Period (days)", group="Liquidity")
useMedianADTV      = input.bool(true, title="Use Median ADTV (robust to block-trade skew)?", tooltip="Off = simple average. Median is highly recommended to prevent volume spikes from skewing the baseline.", group="Liquidity")
minDollarADTV      = input.float(2000000, title="Absolute Illiquidity Floor ($ ADTV)", tooltip="If dollar ADTV falls below this, the asset is flagged illiquid regardless of your position size ratio.", group="Liquidity")

atrLength          = input.int(14, title="ATR Volatility Period", group="Volatility")
volDropThreshold   = input.float(30.0, title="Liquidity Alert: Volume Drop Threshold (%)", minval=1.0, maxval=99.0, group="Volatility")

spreadLookback     = input.int(20, title="Spread Proxy Lookback (days)", group="Exit Cost")
spreadOfRangePct   = input.float(20.0, title="Assumed Spread as % of True Range", tooltip="Estimates round-trip spread as a fraction of the average True Range (accounts for gaps). Calibrate against real Level-1/2 data if available.", minval=1.0, maxval=100.0, group="Exit Cost")

// DYNAMIC ASSET CLASS PRESETS
assetClass         = input.string("Large Cap Equity", title="Asset Class (Impact Calibration)", options=["Large Cap Equity", "Small Cap Equity", "Crypto Major (BTC/ETH)", "Crypto Altcoin", "Forex", "Custom"], group="Exit Cost")
customImpact       = input.float(0.5, title="Custom Impact Coefficient (η)", tooltip="Only used if 'Custom' is selected above.", group="Exit Cost")

impactCoefficient  = switch assetClass
    "Large Cap Equity"       => 0.3
    "Small Cap Equity"       => 0.7
    "Crypto Major (BTC/ETH)" => 0.8
    "Crypto Altcoin"         => 1.5
    "Forex"                  => 0.1
    "Custom"                 => customImpact
    => 0.5 // Fallback

is24_7Market       = input.bool(false, title="24/7 Market (Crypto)? — affects 'Days' label only", group="Display")

// =====================================================================
// 3. MARKET DYNAMICS & VOLATILITY
// =====================================================================
closePrice = close

// --- Robust ADTV: median-based option to resist single block-trade skew ---
adtvSma    = ta.sma(volume, adtvLength)
adtvMedian = ta.percentile_nearest_rank(volume, adtvLength, 50)
adtv       = useMedianADTV ? adtvMedian : adtvSma
dollarADTV = adtv * closePrice

// --- Volatility (ATR-based) ---
atr           = ta.atr(atrLength)
volatilityPct = closePrice > 0 ? (atr / closePrice) * 100.0 : na
normalVolPct  = ta.sma(volatilityPct, adtvLength)

// =====================================================================
// 4. DYNAMIC PARTICIPATION HAIRCUT
// =====================================================================
volatilityScaleFactor = (not na(volatilityPct)) and (not na(normalVolPct)) and volatilityPct > normalVolPct and volatilityPct > 0
     ? (normalVolPct / volatilityPct)
     : 1.0

adjustedParticipationRaw = baseParticipation * volatilityScaleFactor
adjustedParticipation    = math.max(adjustedParticipationRaw, minParticipation)

// =====================================================================
// 5. LIQUIDATION TIME
// =====================================================================
pctRate         = adjustedParticipation / 100.0
maxDailySell    = adtv * pctRate
daysToLiquidate = (not na(maxDailySell)) and maxDailySell > 0 ? myPositionSize / maxDailySell : na

// =====================================================================
// 6. EXIT COST ESTIMATE (Improved with True Range)
// =====================================================================
positionValue = myPositionSize * closePrice

// Spread proxy: Uses True Range (ta.tr) instead of (high-low) to account for overnight/intraday gaps
avgTrueRangePct = ta.sma(ta.tr / closePrice, spreadLookback) * 100.0
estSpreadPct    = avgTrueRangePct * (spreadOfRangePct / 100.0)
estSpreadCost   = positionValue * (estSpreadPct / 100.0)

// Market impact proxy: square-root law (Almgren-Chriss style)
impactRatio   = (not na(adtv)) and adtv > 0 ? myPositionSize / adtv : na
estImpactPct  = (not na(impactRatio)) and impactRatio >= 0 ? impactCoefficient * math.sqrt(impactRatio) * 100.0 : na
estImpactCost = (not na(estImpactPct)) ? positionValue * (estImpactPct / 100.0) : na

estTotalExitCostPct = (not na(estImpactPct)) ? estSpreadPct + estImpactPct : na
estTotalExitCost    = (not na(estImpactCost)) ? estSpreadCost + estImpactCost : na

// =====================================================================
// 7. LIQUIDITY RISK FLAGS
// =====================================================================
isVolumeThin         = (not na(adtv)) and volume < (adtv * (1.0 - (volDropThreshold / 100.0)))
isHighRiskSpread     = (not na(volatilityPct)) and (not na(normalVolPct)) and volatilityPct > (normalVolPct * 1.5)
isBelowAbsoluteFloor = (not na(dollarADTV)) and dollarADTV < minDollarADTV
isOverMaxDays        = (not na(daysToLiquidate)) and daysToLiquidate > maxAcceptableDays

// =====================================================================
// 8. VISUAL RENDERING
// =====================================================================
plot(daysToLiquidate, title="Days to Liquidate", color=color.orange, linewidth=2)
hline(1.0, "1 Day Horizon", color=color.gray, linestyle=hline.style_dashed)
hline(maxAcceptableDays, "Max Acceptable Horizon", color=color.red, linestyle=hline.style_dotted)

// Priority-based background coloring: Absolute Floor > Max Days > Spread Blowout > Thin Liquidity
bgColor = isBelowAbsoluteFloor ? color.new(color.purple, 75) :
          isOverMaxDays        ? color.new(color.red, 80) :
          isHighRiskSpread     ? color.new(color.orange, 85) :
          isVolumeThin         ? color.new(color.yellow, 90) : na
bgcolor(bgColor, title="Risk Level Background")

// =====================================================================
// 9. DASHBOARD (Split Tables)
// =====================================================================
var table coreTable = table.new(position = position.top_right, columns = 2, rows = 7, bgcolor = color.new(color.black, 15), border_width = 1)
var table costTable = table.new(position = position.bottom_right, columns = 2, rows = 4, bgcolor = color.new(color.black, 15), border_width = 1)

if barstate.islast
    // --- Core liquidity table ---
    table.cell(coreTable, 0, 0, "ADTV (" + str.tostring(adtvLength) + "d):", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 0, na(adtv) ? "N/A" : str.tostring(math.round(adtv)), text_color=color.yellow, text_size=size.small)

    table.cell(coreTable, 0, 1, "$ ADTV:", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 1, na(dollarADTV) ? "N/A" : "$" + str.tostring(math.round(dollarADTV)), text_color=isBelowAbsoluteFloor ? color.fuchsia : color.yellow, text_size=size.small)

    table.cell(coreTable, 0, 2, "Adj. Exit Rate:", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 2, str.tostring(adjustedParticipation, "#.##") + "%", text_color=color.aqua, text_size=size.small)

    table.cell(coreTable, 0, 3, "Daily Sell Amount:", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 3, na(maxDailySell) ? "N/A" : str.tostring(math.round(maxDailySell)), text_color=color.lime, text_size=size.small)

    string daysLabel = is24_7Market ? "Days to Exit:" : "Trading Days to Exit:"
    table.cell(coreTable, 0, 4, daysLabel, text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 4, na(daysToLiquidate) ? "N/A" : str.tostring(daysToLiquidate, "#.##") + " Days", text_color=isOverMaxDays ? color.red : color.orange, text_size=size.small)
    
    table.cell(coreTable, 0, 5, "Max Horizon:", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 5, str.tostring(maxAcceptableDays) + " Days", text_color=color.gray, text_size=size.small)

    // Dynamic Risk Status
    string ratioFlag = isBelowAbsoluteFloor ? "🟣 BELOW ABS. FLOOR" : 
                       isOverMaxDays        ? "⏳ EXCEEDS MAX DAYS" : 
                       isHighRiskSpread     ? "💥 SPREAD BLOWOUT" : 
                       isVolumeThin         ? "⚠️ THIN LIQUIDITY" : "✅ NORMAL"
    color ratioColor = isBelowAbsoluteFloor ? color.purple : 
                       isOverMaxDays        ? color.red : 
                       isHighRiskSpread     ? color.orange : 
                       isVolumeThin         ? color.yellow : color.green
    table.cell(coreTable, 0, 6, "Liquidity Risk:", text_color=color.white, text_size=size.small)
    table.cell(coreTable, 1, 6, ratioFlag, text_color=ratioColor, text_size=size.small)

    // --- Exit cost table ---
    table.cell(costTable, 0, 0, "Est. Spread Cost:", text_color=color.white, text_size=size.small)
    table.cell(costTable, 1, 0, na(estSpreadCost) ? "N/A" : "$" + str.tostring(math.round(estSpreadCost)) + " (" + str.tostring(estSpreadPct, "#.##") + "%)", text_color=color.silver, text_size=size.small)

    table.cell(costTable, 0, 1, "Est. Market Impact:", text_color=color.white, text_size=size.small)
    table.cell(costTable, 1, 1, na(estImpactCost) ? "N/A" : "$" + str.tostring(math.round(estImpactCost)) + " (" + str.tostring(estImpactPct, "#.##") + "%)", text_color=color.silver, text_size=size.small)

    table.cell(costTable, 0, 2, "Est. Total Exit Cost:", text_color=color.white, text_size=size.small)
    table.cell(costTable, 1, 2, na(estTotalExitCost) ? "N/A" : "$" + str.tostring(math.round(estTotalExitCost)) + " (" + str.tostring(estTotalExitCostPct, "#.##") + "%)", text_color=color.orange, text_size=size.small)

    table.cell(costTable, 0, 3, "Note:", text_color=color.gray, text_size=size.tiny)
    table.cell(costTable, 1, 3, "Single-symbol proxy — no cross-broker view", text_color=color.gray, text_size=size.tiny)

// =====================================================================
// 10. ALERTS (Enhanced with TradingView Placeholders)
// =====================================================================
alertcondition(isBelowAbsoluteFloor, title="🚨 Below Absolute Liquidity Floor", 
     message="LIQUIDITY ALERT: {{ticker}} Dollar ADTV has fallen below your configured illiquidity floor. Current Price: {{close}}")

alertcondition(isOverMaxDays, title="⏳ Exceeds Max Acceptable Days", 
     message="TIME RISK ALERT: {{ticker}} now requires more than your max acceptable days to liquidate. Current Price: {{close}}")

alertcondition(isHighRiskSpread, title="💥 Spread Blowout Risk", 
     message="VOLATILITY ALERT: {{ticker}} volatility is running >50% above normal. Expect severe slippage. Current Price: {{close}}")

alertcondition(isVolumeThin, title="⚠️ Thin Liquidity Warning", 
     message="VOLUME ALERT: {{ticker}} daily volume is well below the ADTV benchmark. Current Price: {{close}}")
