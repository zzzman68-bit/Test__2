//@version=5
indicator("🟥🟥🟥🟥 Days to Liquidate & Liquidity Stress Exit Tracker v7", overlay=false, max_labels_count=500)

// =====================================================================
// 1. POSITION & PARTICIPATION
// =====================================================================

groupPosition = "Position & Participation"

myPositionSize = input.float(
     100000,
     title="Your Position Size (Shares/Tokens)",
     minval=1,
     group=groupPosition)

baseParticipation = input.float(
     10.0,
     title="Base Daily Participation Rate (%)",
     minval=0.1,
     maxval=100.0,
     group=groupPosition)

minParticipation = input.float(
     0.5,
     title="Minimum Participation Rate Floor (%)",
     minval=0.01,
     maxval=100.0,
     group=groupPosition)


// =====================================================================
// 2. LIQUIDITY
// =====================================================================

groupLiquidity = "Liquidity"

adtvLength = input.int(
     20,
     title="ADTV Benchmark Period (days)",
     minval=2,
     group=groupLiquidity)

useMedianADTV = input.bool(
     true,
     title="Use Median ADTV",
     tooltip="Median is more resistant to abnormal block-trade and event-volume spikes.",
     group=groupLiquidity)

minDollarADTV = input.float(
     2000000,
     title="Absolute Minimum Dollar ADTV",
     minval=0,
     group=groupLiquidity)


// =====================================================================
// 3. VOLATILITY
// =====================================================================

groupVolatility = "Volatility"

atrLength = input.int(
     14,
     title="ATR Volatility Period",
     minval=1,
     group=groupVolatility)

volatilityLookback = input.int(
     20,
     title="Normal Volatility Benchmark",
     minval=2,
     group=groupVolatility)

volDropThreshold = input.float(
     30.0,
     title="Volume Drop Warning (%)",
     minval=1.0,
     maxval=99.0,
     group=groupVolatility)

highVolMultiplier = input.float(
     1.5,
     title="High Volatility Trigger (x Normal)",
     minval=1.0,
     group=groupVolatility)


// =====================================================================
// 4. EXECUTION COST
// =====================================================================

groupCost = "Execution Cost"

spreadLookback = input.int(
     20,
     title="Spread Proxy Lookback",
     minval=2,
     group=groupCost)

spreadOfRangePct = input.float(
     20.0,
     title="Range-Based Spread Proxy (% of Daily Range)",
     minval=1.0,
     maxval=100.0,
     tooltip="This is a proxy, not an actual bid/ask spread.",
     group=groupCost)

impactCoefficient = input.float(
     0.10,
     title="Base Market Impact Coefficient",
     minval=0.0,
     step=0.01,
     tooltip="Square-root impact coefficient. Calibrate against your own execution data where possible.",
     group=groupCost)

impactLowMultiplier = input.float(
     0.50,
     title="Low Impact Scenario Multiplier",
     minval=0.0,
     group=groupCost)

impactHighMultiplier = input.float(
     2.00,
     title="High Impact Scenario Multiplier",
     minval=0.0,
     group=groupCost)


// =====================================================================
// 5. LIQUIDITY STRESS
// =====================================================================

groupStress = "Liquidity Stress Testing"

moderateVolumeFactor = input.float(
     0.75,
     title="Moderate Stress: Volume Factor",
     minval=0.05,
     maxval=1.0,
     group=groupStress)

highVolumeFactor = input.float(
     0.50,
     title="High Stress: Volume Factor",
     minval=0.05,
     maxval=1.0,
     group=groupStress)

severeVolumeFactor = input.float(
     0.25,
     title="Severe Stress: Volume Factor",
     minval=0.01,
     maxval=1.0,
     group=groupStress)

moderateParticipationFactor = input.float(
     0.80,
     title="Moderate Stress: Participation Factor",
     minval=0.05,
     maxval=1.0,
     group=groupStress)

highParticipationFactor = input.float(
     0.50,
     title="High Stress: Participation Factor",
     minval=0.05,
     maxval=1.0,
     group=groupStress)

severeParticipationFactor = input.float(
     0.30,
     title="Severe Stress: Participation Factor",
     minval=0.01,
     maxval=1.0,
     group=groupStress)


// =====================================================================
// 6. DISPLAY
// =====================================================================

groupDisplay = "Display"

is24_7Market = input.bool(
     false,
     title="24/7 Market (Crypto)?",
     tooltip="Only changes the Days/Trading Days label.",
     group=groupDisplay)

showStressPlots = input.bool(
     true,
     title="Show Stress Liquidation Plots",
     group=groupDisplay)


// =====================================================================
// 7. MARKET DATA
// =====================================================================

closePrice = close

atr = ta.atr(atrLength)

volatilityPct =
     closePrice > 0 ?
     (atr / closePrice) * 100.0 :
     na

normalVolPct =
     ta.sma(volatilityPct, volatilityLookback)


// =====================================================================
// 8. VOLUME / ADTV
// =====================================================================

// Share/token volume benchmark
adtvSMA = ta.sma(volume, adtvLength)

adtvMedian =
     ta.percentile_nearest_rank(
          volume,
          adtvLength,
          50)

adtv =
     useMedianADTV ?
     adtvMedian :
     adtvSMA


// Dollar-volume benchmark
dollarVolume = volume * closePrice

dollarADTVSMA =
     ta.sma(dollarVolume, adtvLength)

dollarADTVMedian =
     ta.percentile_nearest_rank(
          dollarVolume,
          adtvLength,
          50)

dollarADTV =
     useMedianADTV ?
     dollarADTVMedian :
     dollarADTVSMA


// =====================================================================
// 9. POSITION METRICS
// =====================================================================

positionValue =
     myPositionSize * closePrice

positionToADTV =
     not na(adtv) and adtv > 0 ?
     (myPositionSize / adtv) * 100.0 :
     na

positionToDollarADTV =
     not na(dollarADTV) and dollarADTV > 0 ?
     (positionValue / dollarADTV) * 100.0 :
     na


// =====================================================================
// 10. DYNAMIC VOLATILITY PARTICIPATION
// =====================================================================

volatilityScaleFactor =
     not na(volatilityPct) and
     not na(normalVolPct) and
     volatilityPct > normalVolPct and
     volatilityPct > 0 ?
     normalVolPct / volatilityPct :
     1.0

adjustedParticipationRaw =
     baseParticipation * volatilityScaleFactor

adjustedParticipation =
     math.max(
          adjustedParticipationRaw,
          minParticipation)


// =====================================================================
// 11. NORMAL LIQUIDATION
// =====================================================================

normalDailySell =
     not na(adtv) ?
     adtv * adjustedParticipation / 100.0 :
     na

daysToLiquidate =
     not na(normalDailySell) and normalDailySell > 0 ?
     myPositionSize / normalDailySell :
     na


// =====================================================================
// 12. RANGE-BASED SPREAD PROXY
// =====================================================================

dailyRangePct =
     closePrice > 0 ?
     ((high - low) / closePrice) * 100.0 :
     na

avgRangePct =
     ta.sma(
          dailyRangePct,
          spreadLookback)

spreadProxyPct =
     not na(avgRangePct) ?
     avgRangePct * (spreadOfRangePct / 100.0) :
     na

spreadCost =
     not na(spreadProxyPct) ?
     positionValue * spreadProxyPct / 100.0 :
     na


// =====================================================================
// 13. BASE MARKET IMPACT
// =====================================================================

// Use dollar value rather than raw shares/tokens.
// This makes the impact ratio more meaningful across different prices.

impactRatio =
     not na(dollarADTV) and dollarADTV > 0 ?
     positionValue / dollarADTV :
     na

baseImpactPct =
     not na(impactRatio) ?
     impactCoefficient * math.sqrt(impactRatio) * 100.0 :
     na

lowImpactPct =
     not na(impactRatio) ?
     impactCoefficient *
     impactLowMultiplier *
     math.sqrt(impactRatio) *
     100.0 :
     na

highImpactPct =
     not na(impactRatio) ?
     impactCoefficient *
     impactHighMultiplier *
     math.sqrt(impactRatio) *
     100.0 :
     na


// =====================================================================
// 14. EXIT COST SCENARIOS
// =====================================================================

normalExitCostPct =
     not na(spreadProxyPct) and
     not na(baseImpactPct) ?
     spreadProxyPct + baseImpactPct :
     na

lowExitCostPct =
     not na(spreadProxyPct) and
     not na(lowImpactPct) ?
     spreadProxyPct + lowImpactPct :
     na

highExitCostPct =
     not na(spreadProxyPct) and
     not na(highImpactPct) ?
     spreadProxyPct + highImpactPct :
     na

normalExitCost =
     not na(normalExitCostPct) ?
     positionValue * normalExitCostPct / 100.0 :
     na

lowExitCost =
     not na(lowExitCostPct) ?
     positionValue * lowExitCostPct / 100.0 :
     na

highExitCost =
     not na(highExitCostPct) ?
     positionValue * highExitCostPct / 100.0 :
     na


// =====================================================================
// 15. LIQUIDITY STRESS ENGINE
// =====================================================================

// Moderate
moderateADTV =
     adtv * moderateVolumeFactor

moderateParticipation =
     math.max(
          adjustedParticipation *
          moderateParticipationFactor,
          minParticipation)

moderateDailySell =
     moderateADTV *
     moderateParticipation / 100.0

moderateDays =
     moderateDailySell > 0 ?
     myPositionSize / moderateDailySell :
     na


// High
highADTV =
     adtv * highVolumeFactor

highParticipation =
     math.max(
          adjustedParticipation *
          highParticipationFactor,
          minParticipation)

highDailySell =
     highADTV *
     highParticipation / 100.0

highDays =
     highDailySell > 0 ?
     myPositionSize / highDailySell :
     na


// Severe
severeADTV =
     adtv * severeVolumeFactor

severeParticipation =
     math.max(
          adjustedParticipation *
          severeParticipationFactor,
          minParticipation)

severeDailySell =
     severeADTV *
     severeParticipation / 100.0

severeDays =
     severeDailySell > 0 ?
     myPositionSize / severeDailySell :
     na


// =====================================================================
// 16. LIQUIDITY RISK FLAGS
// =====================================================================

isVolumeThin =
     not na(adtv) and
     volume < adtv *
     (1.0 - volDropThreshold / 100.0)

isHighVolatility =
     not na(volatilityPct) and
     not na(normalVolPct) and
     volatilityPct >
     normalVolPct * highVolMultiplier

isBelowAbsoluteFloor =
     not na(dollarADTV) and
     dollarADTV < minDollarADTV


// Position concentration flags

isLargePositionVsADTV =
     not na(positionToADTV) and
     positionToADTV > 10.0

isVeryLargePositionVsADTV =
     not na(positionToADTV) and
     positionToADTV > 25.0


// =====================================================================
// 17. LIQUIDITY STRESS SCORE
// =====================================================================

// Score starts at 0 and increases with independent risk conditions.

float liquidityScore = 0.0

liquidityScore += isBelowAbsoluteFloor ? 30.0 : 0.0
liquidityScore += isHighVolatility ? 25.0 : 0.0
liquidityScore += isVolumeThin ? 15.0 : 0.0
liquidityScore += isLargePositionVsADTV ? 15.0 : 0.0
liquidityScore += isVeryLargePositionVsADTV ? 15.0 : 0.0

liquidityScore := math.min(liquidityScore, 100.0)


// =====================================================================
// 18. RISK STATUS
// =====================================================================

string riskStatus =

     liquidityScore >= 75 ?
     "🔴 SEVERE" :

     liquidityScore >= 50 ?
     "🟠 HIGH" :

     liquidityScore >= 25 ?
     "🟡 MODERATE" :

     "🟢 NORMAL"


// =====================================================================
// 19. VISUALS
// =====================================================================

plot(
     daysToLiquidate,
     title="Normal Days to Liquidate",
     color=color.orange,
     linewidth=2)

plot(
     showStressPlots ? moderateDays : na,
     title="Moderate Stress Days",
     color=color.yellow,
     linewidth=1)

plot(
     showStressPlots ? highDays : na,
     title="High Stress Days",
     color=color.red,
     linewidth=1)

plot(
     showStressPlots ? severeDays : na,
     title="Severe Stress Days",
     color=color.purple,
     linewidth=2)

hline(
     1.0,
     "1 Day Horizon",
     color=color.gray,
     linestyle=hline.style_dashed)


// =====================================================================
// 20. BACKGROUND RISK
// =====================================================================

bgcolor(
     liquidityScore >= 75 ?
     color.new(color.purple, 78) :

     liquidityScore >= 50 ?
     color.new(color.red, 82) :

     liquidityScore >= 25 ?
     color.new(color.orange, 88) :

     na,
     title="Liquidity Stress Background")
